cmake_minimum_required(VERSION 3.22.1)
project(trequests VERSION 1.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
option(BUILD_SHARED_LIBS "Build a loadable Tcl extension" ON)
option(TREQUESTS_TESTING_MODE "Expose libcurl options for unit tests" OFF)
option(TREQUESTS_NETWORK_TESTS "Run tests that contact httpbin.org" OFF)
option(ADDRESS_SANITIZER "Enable address and undefined behavior sanitizers" OFF)
option(TCL_MEM_DEBUG "Track allocations with a memory-debug Tcl build" OFF)
include(CTest)

find_package(CURL 7.87 REQUIRED)
find_package(TclStub 8.6 REQUIRED)
add_library(trequests src/library.c src/common.c src/treqSession.c
    src/treqRequest.c src/treqRequestAuth.c src/treqPool.c)
set_target_properties(trequests PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(trequests PRIVATE ${TCL_INCLUDE_PATH})
target_compile_definitions(trequests PRIVATE TCL_THREADS VERSION=${PROJECT_VERSION})
if(TCL_MEM_DEBUG)
    target_compile_definitions(trequests PRIVATE TCL_MEM_DEBUG)
endif()
target_link_libraries(trequests PRIVATE CURL::libcurl)
if(BUILD_SHARED_LIBS)
    target_compile_definitions(trequests PRIVATE USE_TCL_STUBS)
    target_link_libraries(trequests PRIVATE ${TCL_STUB_LIBRARY})
    configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
else()
    target_compile_definitions(trequests PRIVATE STATIC_BUILD)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(trequests PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ADDRESS_SANITIZER)
    target_compile_options(trequests PRIVATE -g -fsanitize=address,undefined)
    target_link_options(trequests PRIVATE -fsanitize=address,undefined)
endif()
if(TREQUESTS_TESTING_MODE)
    target_compile_definitions(trequests PRIVATE TREQUESTS_TESTING_MODE=1)
    target_sources(trequests PRIVATE src/testing.c)
endif()

if(BUILD_TESTING AND BUILD_SHARED_LIBS)
    if(NOT TCL_TCLSH)
        message(FATAL_ERROR "TCL_TCLSH is required to run tests")
    endif()
    set(test_options -notfile "request.test session.test async.test simple.test")
    if(TREQUESTS_NETWORK_TESTS)
        set(test_options)
    endif()
    add_test(NAME AllUnitTests COMMAND ${TCL_TCLSH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl ${test_options})
    set_tests_properties(AllUnitTests PROPERTIES
        ENVIRONMENT "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 120)
    add_custom_target(shell COMMAND ${CMAKE_COMMAND} -E env
        "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}" ${TCL_TCLSH}
        USES_TERMINAL DEPENDS trequests)
endif()

install(TARGETS trequests
    LIBRARY DESTINATION lib/trequests${PROJECT_VERSION}
    ARCHIVE DESTINATION lib/trequests${PROJECT_VERSION}
    RUNTIME DESTINATION lib/trequests${PROJECT_VERSION})
if(BUILD_SHARED_LIBS)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl
        DESTINATION lib/trequests${PROJECT_VERSION})
endif()
